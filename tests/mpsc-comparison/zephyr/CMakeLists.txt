cmake_minimum_required(VERSION 3.20.0)
if(NOT MPSC_BOARD_DIR OR NOT MPSC_RUST_ARCHIVE OR NOT MPSC_SDK_SOURCE)
  message(FATAL_ERROR "Use bench.py build")
endif()
set(CONF_FILE ${MPSC_BOARD_DIR}/prj.conf)
set(DTC_OVERLAY_FILE ${MPSC_BOARD_DIR}/app.overlay)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(mpsc_comparison C)
if(MPSC_LOADER_SOURCE)
  get_target_property(sources zephyr SOURCES)
  list(LENGTH sources before)
  list(FILTER sources EXCLUDE REGEX "/common/loader[.]c$")
  list(LENGTH sources after)
  math(EXPR expected "${after}+1")
  if(NOT before EQUAL expected)
    message(FATAL_ERROR "Expected one SIMPLE_BOOT loader")
  endif()
  set_property(TARGET zephyr PROPERTY SOURCES "${sources}")
  target_sources(zephyr PRIVATE ${MPSC_LOADER_SOURCE})
endif()
target_sources(app PRIVATE ${MPSC_SDK_SOURCE} ${MPSC_BOARD_DIR}/clock.c ${MPSC_BOARD_DIR}/console_zephyr.c)
if(EXISTS ${MPSC_BOARD_DIR}/bootstrap.c)
  target_sources(app PRIVATE ${MPSC_BOARD_DIR}/bootstrap.c)
endif()
target_link_libraries(app PUBLIC ${MPSC_RUST_ARCHIVE})
